  <script>
    const STORAGE_KEY = "simple-vietnamese-inventory";
    const money = new Intl.NumberFormat("vi-VN", { style: "currency", currency: "VND" });
    const number = new Intl.NumberFormat("vi-VN");

    const sampleData = [
      { code: "LOC-001", name: "Lọc dầu động cơ Toyota", category: "Lọc", location: "A1-03", quantity: 42, minimum: 10, price: 85000 },
      { code: "BRK-220", name: "Má phanh trước Ford Ranger", category: "Phanh", location: "B2-11", quantity: 8, minimum: 12, price: 620000 },
      { code: "SPK-778", name: "Bugi Iridium Honda", category: "Động cơ", location: "C1-02", quantity: 64, minimum: 20, price: 145000 },
      { code: "BAT-550", name: "Ắc quy 12V 55Ah", category: "Điện", location: "D4-01", quantity: 5, minimum: 6, price: 1380000 }
    ];

    let inventory = loadInventory();
    let history = [];

    const elements = {
      searchInput: document.querySelector("#searchInput"),
      inventoryBody: document.querySelector("#inventoryBody"),
      partCodes: document.querySelector("#partCodes"),
      totalParts: document.querySelector("#totalParts"),
      totalQuantity: document.querySelector("#totalQuantity"),
      lowStockCount: document.querySelector("#lowStockCount"),
      stockValue: document.querySelector("#stockValue"),
      form: document.querySelector("#stockForm"),
      type: document.querySelector("#type"),
      code: document.querySelector("#code"),
      name: document.querySelector("#name"),
      category: document.querySelector("#category"),
      location: document.querySelector("#location"),
      quantity: document.querySelector("#quantity"),
      price: document.querySelector("#price"),
      minimum: document.querySelector("#minimum"),
      message: document.querySelector("#message"),
      history: document.querySelector("#history"),
      resetButton: document.querySelector("#resetButton")
    };

    function loadInventory() {
      const saved = localStorage.getItem(STORAGE_KEY);
      return saved ? JSON.parse(saved) : sampleData.map((item) => ({ ...item }));
    }

    function saveInventory() {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(inventory));
    }

    function escapeHtml(value) {
      return String(value).replace(/[&<>"']/g, (char) => ({
        "&": "&amp;",
        "<": "&lt;",
        ">": "&gt;",
        '"': "&quot;",
        "'": "&#039;"
      }[char]));
    }

    function getStatus(item) {
      if (item.quantity <= 0) return { text: "Hết hàng", className: "empty" };
      if (item.quantity <= item.minimum) return { text: "Sắp hết", className: "low" };
      return { text: "Đủ hàng", className: "ok" };
    }

    function getFilteredInventory() {
      const keyword = elements.searchInput.value.trim().toLowerCase();
      if (!keyword) return inventory;
      return inventory.filter((item) =>
        [item.code, item.name, item.category, item.location].some((field) => String(field).toLowerCase().includes(keyword))
      );
    }

    function renderTable() {
      const rows = getFilteredInventory();
      elements.inventoryBody.innerHTML = rows.map((item) => {
        const status = getStatus(item);
        return `
          <tr>
            <td><strong>${escapeHtml(item.code)}</strong></td>
            <td>${escapeHtml(item.name)}</td>
            <td>${escapeHtml(item.category)}</td>
            <td>${escapeHtml(item.location)}</td>
            <td>${number.format(item.quantity)}</td>
            <td>${money.format(item.price)}</td>
            <td><span class="status ${status.className}">${status.text}</span></td>
          </tr>
        `;
      }).join("");

      if (!rows.length) {
        elements.inventoryBody.innerHTML = '<tr><td colspan="7">Không tìm thấy phụ tùng phù hợp.</td></tr>';
      }

      elements.partCodes.innerHTML = inventory.map((item) => `<option value="${escapeHtml(item.code)}">${escapeHtml(item.name)}</option>`).join("");
    }

    function renderStats() {
      elements.totalParts.textContent = number.format(inventory.length);
      elements.totalQuantity.textContent = number.format(inventory.reduce((sum, item) => sum + item.quantity, 0));
      elements.lowStockCount.textContent = number.format(inventory.filter((item) => item.quantity <= item.minimum).length);
      elements.stockValue.textContent = money.format(inventory.reduce((sum, item) => sum + item.quantity * item.price, 0));
    }

    function renderHistory() {
      if (!history.length) {
        elements.history.innerHTML = '<li>Chưa có giao dịch trong phiên này.</li>';
        return;
      }

      elements.history.innerHTML = history.slice(0, 5).map((entry) => `
        <li>
          <strong>${entry.type === "import" ? "Nhập" : "Xuất"} ${number.format(entry.quantity)} - ${escapeHtml(entry.code)}</strong>
          ${escapeHtml(entry.name)} • ${escapeHtml(entry.time)}
        </li>
      `).join("");
    }

    function render() {
      renderTable();
      renderStats();
      renderHistory();
    }

    function fillFormFromCode() {
      const code = elements.code.value.trim().toUpperCase();
      const item = inventory.find((part) => part.code === code);
      if (!item) return;
      elements.name.value = item.name;
      elements.category.value = item.category;
      elements.location.value = item.location;
      elements.price.value = item.price;
      elements.minimum.value = item.minimum;
    }

    elements.form.addEventListener("submit", (event) => {
      event.preventDefault();
      const type = elements.type.value;
      const code = elements.code.value.trim().toUpperCase();
      const name = elements.name.value.trim();
      const category = elements.category.value.trim();
      const location = elements.location.value.trim().toUpperCase();
      const quantity = Number(elements.quantity.value);
      const price = Number(elements.price.value);
      const minimum = Number(elements.minimum.value);
      let item = inventory.find((part) => part.code === code);

      if (type === "export" && (!item || item.quantity < quantity)) {
        elements.message.textContent = "Không đủ hàng để xuất kho.";
        elements.message.style.color = "var(--primary)";
        return;
      }

      if (!item) {
        item = { code, name, category, location, quantity: 0, minimum, price };
        inventory.push(item);
      }

      item.name = name;
      item.category = category;
      item.location = location;
      item.minimum = minimum;
      item.price = price;
      item.quantity += type === "import" ? quantity : -quantity;

      history.unshift({ type, code, name, quantity, time: new Date().toLocaleString("vi-VN") });
      saveInventory();
      render();
      elements.form.reset();
      elements.quantity.value = 1;
      elements.price.value = 0;
      elements.minimum.value = 5;
      elements.message.style.color = "var(--green)";
      elements.message.textContent = type === "import" ? "Đã nhập kho thành công." : "Đã xuất kho thành công.";
    });

    elements.searchInput.addEventListener("input", renderTable);
    elements.code.addEventListener("change", fillFormFromCode);
    elements.resetButton.addEventListener("click", () => {
      inventory = sampleData.map((item) => ({ ...item }));
      history = [];
      saveInventory();
      render();
      elements.message.style.color = "var(--green)";
      elements.message.textContent = "Đã khôi phục dữ liệu mẫu.";
    });

    render();
  </script>
